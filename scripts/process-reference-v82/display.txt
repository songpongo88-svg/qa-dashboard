export function ProcessReferenceDisplay({ value, className = "" }: { value: string; className?: string }) {
  const refs = useMemo(() => parseProcessReferenceListV68(value), [value]);
  const [previewMeta, setPreviewMeta] = useState<ProcessReferenceMeta | null>(null);
  const groups = (() => {
    const grouped = new Map<string, { key: string; processName: string; versionLabel: string; refs: ProcessReferenceMeta[] }>();
    refs.forEach((meta, index) => {
      const fallbackIdentity = [
        meta.processName || "Process",
        meta.versionLabel || "-",
        meta.fileUrl || "",
        meta.fileType || "",
      ].join("||");
      const key = String(meta.versionId || "").trim() || fallbackIdentity || "process-" + index;
      const existing = grouped.get(key);
      if (existing) {
        existing.refs.push(meta);
        return;
      }
      grouped.set(key, {
        key,
        processName: meta.processName || "Process",
        versionLabel: meta.versionLabel || "-",
        refs: [meta],
      });
    });
    return Array.from(grouped.values());
  })();

  if (!refs.length) return <RichTextContent value={value} className={"whitespace-pre-line text-[14px] leading-6 text-slate-800 " + className} />;

  return (
    <div className={className}>
      <div className="space-y-3">
        {groups.map((group) => (
          <section key={group.key} className="rounded-2xl border border-sky-100 bg-sky-50/30 p-3">
            <div className="min-w-0">
              <div className="text-[10px] font-black uppercase tracking-[0.14em] text-slate-500">Process</div>
              <div className="mt-0.5 break-words text-sm font-black text-slate-950" title={group.processName}>{group.processName}</div>
              <div className="mt-1 text-xs font-semibold text-slate-500">Version: {formatProcessVersionV78(group.versionLabel)}</div>
            </div>

            <div className="mt-3 space-y-2">
              {group.refs.map((meta, index) => {
                const displayTitle = meta.step && meta.step !== "ทั้งสไลด์" ? meta.step : (meta.slideTitle || ("Slide " + meta.slideNumber));
                return (
                  <div key={group.key + "-" + meta.slideNumber + "-" + index} className="flex flex-col gap-2 rounded-xl border border-sky-100 bg-white px-3 py-3 shadow-sm sm:flex-row sm:items-center sm:justify-between">
                    <div className="min-w-0">
                      <div className="text-xs font-black text-emerald-700">Slide {meta.slideNumber}</div>
                      <div className="mt-0.5 text-sm font-normal text-emerald-700" title={displayTitle}>{displayTitle}</div>
                    </div>
                    <button type="button" onClick={() => setPreviewMeta({ ...meta, __openTokenV82: Date.now() } as any)} className="shrink-0 rounded-xl border border-sky-200 bg-sky-50 px-3 py-2 text-xs font-black text-[#155B83] shadow-sm transition hover:bg-sky-100">◉ ดูสไลด์</button>
                  </div>
                );
              })}
            </div>
          </section>
        ))}
      </div>
      <div className="mt-2 text-[11px] font-semibold text-slate-500">ใช้ไฟล์จาก Version ที่บันทึกไว้พร้อมผลประเมินเคสนี้</div>
      <ProcessSlideModalV68 meta={previewMeta} items={refs} openToken={Number((previewMeta as any)?.__openTokenV82 || 0)} onNavigate={(next) => setPreviewMeta({ ...next, __openTokenV82: Date.now() } as any)} onClose={() => setPreviewMeta(null)} />
    </div>
  );
}
